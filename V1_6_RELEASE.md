import http from 'node:http';
import { readFileSync, existsSync } from 'node:fs';
import { extname, resolve } from 'node:path';
import { fileURLToPath } from 'node:url';
import { SQLiteStore, YucelyService, OneButtonEngine, AuthService, RateLimiter, bearer, sameUserOrThrow, calculateBudget, buildDaily3, calculateCfo, simulateWhatIf, fixMyDay, AgentOrchestrator, classifySignal, buildVoiceReport, buildFeatureReport, buildOpportunityRadar, validateMatchProfile, rankMutualMatches, createConnection, approveConnection, CommerceEngine, verifyStripeSignature, AsyncCommerceEngine, StripeConnectProvider } from '../../core/src/index.js';

const __dirname = fileURLToPath(new URL('.', import.meta.url));
const webRoot = resolve(__dirname, '../../web');
const mime={'.html':'text/html; charset=utf-8','.css':'text/css; charset=utf-8','.js':'text/javascript; charset=utf-8'};
function staticFile(res,path){const file=resolve(webRoot,path);if(!file.startsWith(webRoot)||!existsSync(file))return false;const data=readFileSync(file);res.writeHead(200,{'content-type':mime[extname(file)]||'application/octet-stream','cache-control':'no-cache','x-content-type-options':'nosniff'});res.end(data);return true;}

export function makeApp({dbPath=process.env.DB_PATH??'/tmp/projeyucely.db', rateMax=120, stripeProvider=null}={}){
 const store=new SQLiteStore(dbPath), service=new YucelyService(store), oneButton=new OneButtonEngine(store,service), agents=new AgentOrchestrator(store,service), commerce=new CommerceEngine({store}), auth=new AuthService(store), limiter=new RateLimiter({max:rateMax});
 const resolvedStripeProvider=stripeProvider??(process.env.STRIPE_SECRET_KEY?new StripeConnectProvider({secretKey:process.env.STRIPE_SECRET_KEY}):null);
 const stripeCommerce=resolvedStripeProvider?new AsyncCommerceEngine({provider:resolvedStripeProvider}):null;
 function json(res,code,body,extra={}){res.writeHead(code,{'content-type':'application/json; charset=utf-8','x-content-type-options':'nosniff','cache-control':'no-store',...extra});res.end(JSON.stringify(body));}
 async function readRaw(req){const chunks=[];let size=0;for await(const c of req){size+=c.length;if(size>256000)throw new Error('BODY_TOO_LARGE');chunks.push(c);}return Buffer.concat(chunks).toString('utf8');}
 async function readBody(req){const raw=await readRaw(req);return raw?JSON.parse(raw):{};}
 const stripeEvents=new Set();
 return http.createServer(async(req,res)=>{const ip=req.socket.remoteAddress??'unknown';const rate=limiter.check(ip);if(!rate.allowed)return json(res,429,{error:'RATE_LIMITED'},{'retry-after':String(Math.ceil(rate.reset_ms/1000))});
  try{const url=new URL(req.url,'http://localhost');
   if(req.method==='POST'&&url.pathname==='/v1/stripe/webhook'){const raw=await readRaw(req);const secret=process.env.STRIPE_WEBHOOK_SECRET;if(!secret)return json(res,503,{error:'STRIPE_WEBHOOK_NOT_CONFIGURED'});const ok=verifyStripeSignature({payload:raw,signatureHeader:req.headers['stripe-signature'],webhookSecret:secret});if(!ok)return json(res,400,{error:'INVALID_STRIPE_SIGNATURE'});const event=JSON.parse(raw);if(stripeEvents.has(event.id))return json(res,200,{received:true,duplicate:true});stripeEvents.add(event.id);store.appendAudit({actor_id:null,action:'STRIPE_WEBHOOK_RECEIVED',resource_type:'STRIPE_EVENT',resource_id:event.id??'unknown',ip,metadata:{type:event.type??'unknown',livemode:!!event.livemode}});return json(res,200,{received:true});}
   const body=['POST','PUT','PATCH'].includes(req.method)?await readBody(req):{};
   if(['GET','HEAD'].includes(req.method)&&url.pathname==='/')return staticFile(res,'index.html');
   if(['GET','HEAD'].includes(req.method)&&url.pathname.startsWith('/app/'))return staticFile(res,url.pathname.slice(5));
   if(req.method==='GET'&&url.pathname==='/health')return json(res,200,{ok:true,service:'projeyucely-api',version:'2.0.0-rc1'});
   if(req.method==='POST'&&url.pathname==='/v1/auth/register'){const out=auth.register(body);store.appendAudit({actor_id:out.user.id,action:'AUTH_REGISTER',resource_type:'USER',resource_id:out.user.id,ip});return json(res,201,out);}
   if(req.method==='POST'&&url.pathname==='/v1/auth/login'){const out=auth.login(body.email,body.password);store.appendAudit({actor_id:out.user.id,action:'AUTH_LOGIN',resource_type:'USER',resource_id:out.user.id,ip});return json(res,200,out);}
   const actor=auth.authenticate(bearer(req)); if(!actor)return json(res,401,{error:'UNAUTHORIZED'});
   let m;
   if(req.method==='GET'&&url.pathname==='/v1/stripe-connect/status'){return json(res,200,{configured:!!stripeCommerce,mode:process.env.STRIPE_SECRET_KEY?.startsWith('sk_live_')?'live':stripeCommerce?'test_or_injected':'disabled'});}
   if(req.method==='POST'&&url.pathname==='/v1/stripe-connect/onboard'){if(!stripeCommerce)return json(res,503,{error:'STRIPE_NOT_CONFIGURED'});const r=await stripeCommerce.onboardWorker({workerId:actor.id,email:actor.email??null,country:body.country??'US',refreshUrl:body.refresh_url,returnUrl:body.return_url,idempotencyKey:body.idempotency_key});store.appendAudit({actor_id:actor.id,action:'STRIPE_CONNECT_ONBOARD',resource_type:'CONNECTED_ACCOUNT',resource_id:r.account_id,ip,metadata:{status:r.status}});return json(res,201,r);}
   if(req.method==='POST'&&url.pathname==='/v1/stripe-connect/refresh'){if(!stripeCommerce)return json(res,503,{error:'STRIPE_NOT_CONFIGURED'});const r=await stripeCommerce.refreshWorkerAccount(actor.id);store.appendAudit({actor_id:actor.id,action:'STRIPE_CONNECT_REFRESH',resource_type:'CONNECTED_ACCOUNT',resource_id:r.account_id,ip,metadata:{status:r.status}});return json(res,200,r);}
   if(req.method==='POST'&&url.pathname==='/v1/stripe-connect/jobs/fund'){if(!stripeCommerce)return json(res,503,{error:'STRIPE_NOT_CONFIGURED'});const j=await stripeCommerce.fundJob({requesterId:actor.id,workerId:body.worker_id,amount:body.amount,currency:body.currency??'usd',idempotencyKey:body.idempotency_key});store.appendAudit({actor_id:actor.id,action:'STRIPE_TEST_JOB_FUNDED',resource_type:'COMMERCE_JOB',resource_id:j.id,ip,metadata:{amount:j.amount}});return json(res,201,j);}
   m=url.pathname.match(/^\/v1\/stripe-connect\/jobs\/([^/]+)\/complete$/); if(req.method==='POST'&&m){if(!stripeCommerce)return json(res,503,{error:'STRIPE_NOT_CONFIGURED'});const j=stripeCommerce.jobs.get(m[1]);if(!j)throw new Error('JOB_NOT_FOUND');if(actor.id!==j.requester_id&&process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');const done=await stripeCommerce.completeJob(m[1],{idempotencyKey:body.idempotency_key});store.appendAudit({actor_id:actor.id,action:'STRIPE_TEST_JOB_COMPLETE',resource_type:'COMMERCE_JOB',resource_id:done.id,ip,metadata:{transfer_id:done.transfer_id}});return json(res,200,done);}
   if(req.method==='POST'&&url.pathname==='/v1/one-button'){if(!body.text||!String(body.text).trim())throw new Error('TEXT_REQUIRED');const r=oneButton.execute(actor.id,String(body.text),body.context??{});store.appendAudit({actor_id:actor.id,action:'ONE_BUTTON_EXECUTE',resource_type:'WORKFLOW',resource_id:r.workflow.id,ip,metadata:{intent:r.workflow.intent.primary_intent,status:r.result.status}});store.addFeatureEvent({user_id:actor.id,feature:'ONE_BUTTON',status:r.result.status==='BLOCKED'?'FAILURE':'SUCCESS',metadata:{intent:r.workflow.intent.primary_intent}});return json(res,200,r);}
   m=url.pathname.match(/^\/v1\/users\/([^/]+)\/availability$/); if(req.method==='POST'&&m){sameUserOrThrow(actor,m[1]);const r=service.addAvailability(m[1],body);store.appendAudit({actor_id:actor.id,action:'AVAILABILITY_CREATE',resource_type:'AVAILABILITY',resource_id:r.id,ip});return json(res,201,r);}
   m=url.pathname.match(/^\/v1\/users\/([^/]+)\/needs$/); if(req.method==='POST'&&m){sameUserOrThrow(actor,m[1]);const r=service.postNeed(m[1],body);store.appendAudit({actor_id:actor.id,action:'NEED_CREATE',resource_type:'NEED',resource_id:r.id,ip});return json(res,201,r);}
   m=url.pathname.match(/^\/v1\/users\/([^/]+)\/opportunities$/); if(req.method==='POST'&&m){sameUserOrThrow(actor,m[1]);const r=service.postOpportunity(m[1],body);store.appendAudit({actor_id:actor.id,action:'OPPORTUNITY_CREATE',resource_type:'OPPORTUNITY',resource_id:r.id,ip});return json(res,201,r);}
   m=url.pathname.match(/^\/v1\/needs\/([^/]+)\/matches$/); if(req.method==='GET'&&m)return json(res,200,{matches:service.findMatches(m[1])});
   m=url.pathname.match(/^\/v1\/users\/([^/]+)\/money-mission$/); if(req.method==='POST'&&m){sameUserOrThrow(actor,m[1]);return json(res,200,service.buildMoneyMissionForUser(m[1],body.target_amount));}

   if(req.method==='POST'&&url.pathname==='/v1/agents/tasks'){const t=agents.createTask(actor.id,body);store.appendAudit({actor_id:actor.id,action:'AGENT_TASK_CREATE',resource_type:'AGENT_TASK',resource_id:t.id,ip,metadata:{agent_action:t.action,status:t.status}});return json(res,t.status==='NEEDS_APPROVAL'?202:200,t);}
   m=url.pathname.match(/^\/v1\/agents\/tasks\/([^/]+)\/approve$/); if(req.method==='POST'&&m){const t=agents.approveTask(actor.id,m[1]);store.appendAudit({actor_id:actor.id,action:'AGENT_TASK_APPROVE',resource_type:'AGENT_TASK',resource_id:t.id,ip,metadata:{agent_action:t.action,status:t.status}});return json(res,200,t);}
   if(req.method==='GET'&&url.pathname==='/v1/agents/tasks')return json(res,200,{tasks:store.listUserAgentTasks(actor.id,50)});
   if(req.method==='GET'&&url.pathname==='/v1/me')return json(res,200,{user:actor});
   if(req.method==='GET'&&url.pathname==='/v1/dashboard')return json(res,200,service.getDashboard(actor.id));

   if(req.method==='POST'&&url.pathname==='/v1/commerce/jobs/fund'){
    if(!body.worker_id)throw new Error('WORKER_REQUIRED');
    if(!body.idempotency_key)throw new Error('IDEMPOTENCY_REQUIRED');
    const job=commerce.fundJob({requesterId:actor.id,workerId:body.worker_id,amount:body.amount,currency:body.currency??'usd',opportunityId:body.opportunity_id??null,needId:body.need_id??null,idempotencyKey:body.idempotency_key});
    store.appendAudit({actor_id:actor.id,action:'JOB_FUNDED',resource_type:'COMMERCE_JOB',resource_id:job.id,ip,metadata:{amount:job.amount,currency:job.currency,payment_id:job.payment_id}});
    return json(res,201,job);
   }
   m=url.pathname.match(/^\/v1\/commerce\/jobs\/([^/]+)$/); if(req.method==='GET'&&m){const job=commerce.getJob(m[1]);if(!job)throw new Error('JOB_NOT_FOUND');if(actor.id!==job.requester_id&&actor.id!==job.worker_id&&process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');return json(res,200,job);}
   m=url.pathname.match(/^\/v1\/commerce\/jobs\/([^/]+)\/complete$/); if(req.method==='POST'&&m){const current=commerce.getJob(m[1]);if(!current)throw new Error('JOB_NOT_FOUND');if(actor.id!==current.requester_id&&process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');if(!body.idempotency_key)throw new Error('IDEMPOTENCY_REQUIRED');const job=commerce.completeJob(m[1],{idempotencyKey:body.idempotency_key});store.appendAudit({actor_id:actor.id,action:'JOB_COMPLETED_AND_PAID',resource_type:'COMMERCE_JOB',resource_id:job.id,ip,metadata:{payout_amount:job.payout_amount,platform_fee:job.platform_fee,tax_reserve:job.tax_reserve,safety_reserve:job.safety_reserve}});return json(res,200,job);}
   m=url.pathname.match(/^\/v1\/commerce\/jobs\/([^/]+)\/refund$/); if(req.method==='POST'&&m){const current=commerce.getJob(m[1]);if(!current)throw new Error('JOB_NOT_FOUND');if(actor.id!==current.requester_id&&process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');const job=commerce.refundJob(m[1]);store.appendAudit({actor_id:actor.id,action:'JOB_REFUNDED',resource_type:'COMMERCE_JOB',resource_id:job.id,ip});return json(res,200,job);}
   if(req.method==='GET'&&url.pathname==='/v1/commerce/transactions')return json(res,200,{transactions:commerce.listTransactions({userId:actor.id,limit:Number(url.searchParams.get('limit')??100)})});
   m=url.pathname.match(/^\/v1\/commerce\/jobs\/([^/]+)\/disputes$/); if(req.method==='POST'&&m){const current=commerce.getJob(m[1]);if(!current)throw new Error('JOB_NOT_FOUND');if(actor.id!==current.requester_id&&actor.id!==current.worker_id&&process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');if(!body.idempotency_key)throw new Error('IDEMPOTENCY_REQUIRED');const d=commerce.openDispute(m[1],{openedBy:actor.id,reason:body.reason??'other',amount:body.amount??null,idempotencyKey:body.idempotency_key});store.appendAudit({actor_id:actor.id,action:'DISPUTE_OPENED',resource_type:'COMMERCE_JOB',resource_id:m[1],ip,metadata:{dispute_id:d.id,reason:d.reason}});return json(res,201,d);}
   m=url.pathname.match(/^\/v1\/admin\/commerce\/disputes\/([^/]+)\/resolve$/); if(req.method==='POST'&&m){if(process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');const d=commerce.resolveDispute(m[1],{resolution:body.resolution??'REJECT',note:body.note??null});store.appendAudit({actor_id:actor.id,action:'DISPUTE_RESOLVED',resource_type:'DISPUTE',resource_id:d.id,ip,metadata:{status:d.status}});return json(res,200,d);}
   m=url.pathname.match(/^\/v1\/admin\/commerce\/jobs\/([^/]+)\/reconcile-payout$/); if(req.method==='POST'&&m){if(process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');const j=commerce.reconcilePayout(m[1],{idempotencyKey:body.idempotency_key??null});store.appendAudit({actor_id:actor.id,action:'PAYOUT_RECONCILE',resource_type:'COMMERCE_JOB',resource_id:j.id,ip,metadata:{status:j.status,payout_attempts:j.payout_attempts}});return json(res,200,j);}
   if(req.method==='GET'&&url.pathname==='/v1/admin/commerce/reconciliation'){if(process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');return json(res,200,{report:commerce.reconciliationReport()});}
   if(req.method==='POST'&&url.pathname==='/v1/admin/commerce/finance-snapshot'){if(process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');const snapshot=commerce.financialSnapshot({activeUsers:body.active_users??0,operatingCosts:body.operating_costs??0,refunds:body.refunds??0,fraudLosses:body.fraud_losses??0,taxRate:body.tax_rate??0.25});store.appendAudit({actor_id:actor.id,action:'COMMERCE_FINANCE_SNAPSHOT',resource_type:'FINANCE',resource_id:'commerce-current',ip,metadata:{growth_gate:snapshot.cfo.growth_gate}});return json(res,200,{snapshot});}
   if(req.method==='POST'&&url.pathname==='/v1/mutual-match/profile'){const profile=validateMatchProfile({...body,user_id:actor.id});store.saveMatchProfile(profile);store.appendAudit({actor_id:actor.id,action:'MATCH_PROFILE_SAVE',resource_type:'MATCH_PROFILE',resource_id:profile.id,ip,metadata:{type:profile.type}});store.addFeatureEvent({user_id:actor.id,feature:'MUTUAL_MATCH_PROFILE',status:'SUCCESS',metadata:{type:profile.type}});return json(res,201,profile);}
   if(req.method==='GET'&&url.pathname==='/v1/mutual-match/matches'){const type=String(url.searchParams.get('type')??'').toUpperCase();const profile=store.getUserMatchProfile(actor.id,type);if(!profile)throw new Error('MATCH_PROFILE_NOT_FOUND');const matches=rankMutualMatches(profile,store.listMatchProfiles(type));return json(res,200,{profile_id:profile.id,type,matches});}
   if(req.method==='POST'&&url.pathname==='/v1/mutual-match/connections'){const type=String(body.type??'').toUpperCase();const mine=store.getUserMatchProfile(actor.id,type);const other=store.getMatchProfile(body.other_profile_id);if(!mine||!other)throw new Error('MATCH_PROFILE_NOT_FOUND');const ranked=rankMutualMatches(mine,[other]);if(!ranked.length)throw new Error('MATCH_NOT_ELIGIBLE');const existing=store.listUserMatchConnections(actor.id).find(c=>c.match_id===ranked[0].id);const c=existing??createConnection(ranked[0],actor.id,other.user_id);store.saveMatchConnection(c);store.appendAudit({actor_id:actor.id,action:'MATCH_CONNECTION_CREATE',resource_type:'MATCH_CONNECTION',resource_id:c.id,ip,metadata:{type:c.type}});return json(res,201,c);}

   if(req.method==='POST'&&url.pathname==='/v1/admin/cfo'){if(process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');const cfo=calculateCfo(body);store.appendAudit({actor_id:actor.id,action:'CFO_CALCULATE',resource_type:'FINANCE',resource_id:'current',ip,metadata:{growth_gate:cfo.growth_gate}});return json(res,200,{cfo});}
   if(req.method==='GET'&&url.pathname==='/v1/admin/audit'){if(process.env.ADMIN_USER_ID!==actor.id)throw new Error('FORBIDDEN');return json(res,200,{events:store.listAudit(100)});}
   return json(res,404,{error:'NOT_FOUND'});
  }catch(error){const map={USER_NOT_FOUND:404,NEED_NOT_FOUND:404,FORBIDDEN:403,EMAIL_EXISTS:409,INVALID_CREDENTIALS:401,WEAK_PASSWORD:400,EMAIL_REQUIRED:400,BODY_TOO_LARGE:413,TEXT_REQUIRED:400,TARGET_AMOUNT_REQUIRED:400,ACTION_REQUIRED:400,FEATURE_REQUIRED:400,AGENT_TASK_NOT_FOUND:404,MATCH_PROFILE_NOT_FOUND:404,MATCH_CONNECTION_NOT_FOUND:404,MATCH_NOT_ELIGIBLE:409,INVALID_MATCH_TYPE:400,MATCH_OPT_IN_REQUIRED:400,JOB_NOT_FOUND:404,WORKER_REQUIRED:400,PARTIES_REQUIRED:400,IDEMPOTENCY_REQUIRED:400,INVALID_JOB_STATE:409,PAYMENT_NOT_CAPTURED:402,INVALID_AMOUNT:400,STRIPE_SECRET_KEY_REQUIRED:503,STRIPE_NOT_CONFIGURED:503,CONNECTED_ACCOUNT_NOT_FOUND:404,WORKER_ONBOARDING_REQUIRED:409,ACCOUNT_LINK_FIELDS_REQUIRED:400};return json(res,map[error.message]??400,{error:error.message});}
 });
